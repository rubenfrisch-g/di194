import re
import string
import nltk
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer

stop_words = set(stopwords.words("english"))
lemmatizer = WordNetLemmatizer()


def preprocess_text(text: str) -> str:
    """Lowercase, tokenize, strip punctuation, drop stopwords, and lemmatize a review."""
    text = text.lower()
    tokens = nltk.word_tokenize(text)
    tokens = [token for token in tokens if token not in string.punctuation]
    tokens = [token for token in tokens if token not in stop_words]
    tokens = [lemmatizer.lemmatize(token) for token in tokens]
    tokens = [lemmatizer.lemmatize(token) for token in tokens]
    cleaned_text = " ".join(tokens)
    return " ".join(tokens)

def plot_word_embeddings(model, words=None):
    if words is None:
        words = model.wv.index_to_key[:20]
    x_coords = []
    y_coords = []
    for word in words:
        vector = model.wv[word]
        x_coords.append(vector[0])
        y_coords.append(vector[1])
    plt.figure(figsize=(10, 8))
    for i, word in enumerate(words):

        plt.text(x_coords[i] + 0.01, y_coords[i] + 0.01, word)

plot : To plot scatter plots when markers are identical in size and

Notes


    

  • The .plot function will be faster for scatterplots where markers
      don't vary in size or color.
    • 

  • Any or all of *x*, *y*, *s*, and *c* may be masked arrays, in which
      case all masks will be combined and only unmasked points will be
plotted.
    • 

  • Fundamentally, scatter works with 1D arrays; *x*, *y*, *s*, and *c*
      may be input as N-D arrays, but within scatter they will be
flattened. The exception is *c*, which will be flattened only if its
size matches the size of *x* and *y*.
